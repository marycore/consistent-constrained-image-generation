% C9 Aggregate Comparison: Requires equal cardinality between two independently filtered subsets.
c9_2prop_left(N, V1') :- N = #count { X : object(X), hasProperty(X, P1', V1'), hasProperty(X, P2', V2'), P1'!=P2'}.
c9_2prop_right(N, V3') :- N = #count { Y : object(Y), hasProperty(Y, P3', V3'), hasProperty(Y, P4', V4'), P4'!=P3'}.
:- c9_2prop_left(N, V1'), c9_2prop_right(M, V3'), N!= M, V1'!=V3'.

%% The following constraint(s) are added to avoid generating scneces that are only vacuously true!
:- #count {X : object(X), hasProperty(X, P1', V1'), hasProperty(X, P2', V2'), P1'!=P2'} = 0.

% C3 Conditional Object, negated literal: every object with P1'=V1'  must have P2'=V2' and P3'=V3' and P4'=V4'. 
:- object(X), hasProperty(X, P1', V1'),  P1'!=P2', P1'!=P3', P1'!=P4', P2'!=P3', P2'!=P4', P3'!=P4', #count{ 1 : hasProperty(X, P2', V2') ; 2 : hasProperty(X, P3', V3') ; 3 : hasProperty(X, P4', V4')} != 3.

%% The following constraint(s) are added to avoid generating scneces that are only vacuously true!
:- #count {X: object(X), hasProperty(X, P1', V1')} = 0.
